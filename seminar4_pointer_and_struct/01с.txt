#include <stdio.h>

int main(void)
{
    // ── 1. Указатель на int ─────────────────────────────────
    int a1 = 10;
    int* p1 = &a1;


    // ── 2. Указатель на float ───────────────────────────────
    float a2 = 1.1;
    float* p2 = &a2;


    // ── 3. Указатель на первый элемент массива ──────────────
    int array[5] = {10, 20, 30, 40, 50};
    int* p3 = &array[0];


    // ── 4. Указатель на символ 'A' в строке ─────────────────
    char str[20] = "Sapere Aude";
    char* p4 = &str[7];       // 'A' находится на позиции 7


    // ── 5. Указатель на указатель ───────────────────────────
    int a5 = 10;
    int* p5 = &a5;
    int** q5 = &p5;


    // ── 6. Удвоить a через p ────────────────────────────────
    int a6 = 10;
    int* p6 = &a6;
    *p6 *= 2;                 // a6 стало 20
    printf("6: %i\n", a6);


    // ── 7. Перевести 't' в 'T' через p ──────────────────────
    char a7 = 't';
    char* p7 = &a7;
    *p7 = *p7 - 32;           // 't' (116) - 32 = 'T' (84)
    printf("7: %c\n", a7);


    // ── 8. Добавить 1 к a[3], используя p на a[0] ───────────
    int a8[5] = {10, 20, 30, 40, 50};
    int* p8 = &a8[0];
    *(p8 + 3) += 1;           // a8[3] стало 41
    printf("8: %i\n", a8[3]);


    // ── 9. p на a[3]; добавить 1 к a[0] ─────────────────────
    int a9[5] = {10, 20, 30, 40, 50};
    int* p9 = &a9[3];
    *(p9 - 3) += 1;           // a9[0] стало 11
    printf("9: %i\n", a9[0]);


    // ── 10. Все буквы строки в верхний регистр ──────────────
    char str10[] = "sapereaude";
    char* p10 = &str10[0];
    for (; *p10; ++p10)
        *p10 -= 32;
    printf("10: %s\n", str10);   // SAPEREAUDE


    // ── 11. Удвоить a через q (указатель на указатель) ──────
    int a11 = 10;
    int* p11 = &a11;
    int** q11 = &p11;
    **q11 *= 2;               // a11 стало 20
    printf("11: %i\n", a11);


    return 0;
}
